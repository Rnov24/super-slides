RUNTIME CODE

DeckEngine: Single-Flight Transition Core

The transition pipeline enforces a single-flight lock. Rapid keypresses coalesce to the final target index without spawning concurrent animations.

// Single-flight transition lock with target-index coalescing
goTo(index, broadcast = true, explicitDir = null) {
  if (index < 0 || index >= this.slides.length) return;
  if (index === this.currentIndex && !this.isTransitioning) return;

  // Coalesce target when a transition is already in flight
  if (this.isTransitioning) { this.targetIndex = index; return; }

  const dir = explicitDir !== null ? explicitDir : (index > this.currentIndex ? 1 : -1);
  this.targetIndex = index;
  this._executeTransition(this.currentIndex, index, dir, broadcast);
}

// Dual completion guards: transitionend + safety timer (duration + 40ms)
_executeTransition(fromIdx, toIdx, dir, broadcast) {
  const fromSlide = this.slides[fromIdx];
  const toSlide   = this.slides[toIdx];
  const cut = window.matchMedia('(prefers-reduced-motion: reduce)').matches
            || fromSlide.dataset.transition === 'cut'
            || toSlide.dataset.transition   === 'cut';

  this.isTransitioning = true;
  this.viewport.classList.add('is-transitioning'); // pointer-events: none

  if (cut) { /* instant cut, no will-change */
    fromSlide.classList.remove('slide-current');
    toSlide.classList.add('slide-current');
    this._finalize(toIdx, broadcast); return;
  }

  toSlide.style.transform = `translate3d(${dir * 100}%, 0, 0)`;
  toSlide.classList.add('slide-incoming');
  toSlide.getBoundingClientRect(); // force reflow
  fromSlide.classList.add(dir > 0 ? 'slide-exit-left' : 'slide-exit-right');
  toSlide.classList.add('slide-enter-active');

  let resolved = false;
  const done = () => {
    if (resolved) return; resolved = true;
    clearTimeout(this.safetyTimer);
    toSlide.removeEventListener('transitionend', done);
    this._cleanupTransition(fromSlide, toSlide);
    this._finalize(toIdx, broadcast);
  };
  toSlide.addEventListener('transitionend', done, { once: true });
  this.safetyTimer = setTimeout(done, this.duration + 40); // safety timer
}
